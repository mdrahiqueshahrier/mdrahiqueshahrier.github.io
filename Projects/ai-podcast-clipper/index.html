import { FaceTracker, WhisperTranscriber } from '@clipflow/core';
import { FFmpegRenderer } from '@clipflow/ffmpeg-nvenc';

export async function processPodcastClip(clipMeta) {
  // 1. Transcribe audio with word-level precision
  const transcript = await WhisperTranscriber.transcribe({
    input: clipMeta.audioTrack,
    model: 'large-v3',
    device: 'cuda'
  });

  // 2. Dynamic 9:16 optical face tracking matrix
  const cropBox = await FaceTracker.computeKeyframeTrajectory({
    video: clipMeta.videoTrack,
    aspect: '9:16',
    smoothingFrames: 30
  });

  // 3. Burn animated karaoke captions via GPU
  return await FFmpegRenderer.burnAndExport({
    crop: cropBox,
    captions: transcript.words,
    preset: 'hormozi-glow'
  });
}
